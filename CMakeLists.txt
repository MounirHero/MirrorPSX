cmake_minimum_required(VERSION 3.22)
project(MirrorPSX VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MIRROR_PS5 "Build native PS5 application through OpenProspero" OFF)
option(MIRROR_VALIDATE "Enable desktop Vulkan validation layers" OFF)
if(MIRROR_PS5)
    include(cmake/ps5.cmake)
    return()
endif()
find_package(Vulkan REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
pkg_check_modules(MEDIA REQUIRED IMPORTED_TARGET libavformat libswresample libxml-2.0)
pkg_check_modules(SOUND REQUIRED sdl2)
set(NO_MARCH_NATIVE ON)
add_subdirectory(third_party/uxplay/lib/mdnsd)
add_subdirectory(third_party/uxplay/lib/llhttp)
add_subdirectory(third_party/uxplay/lib/playfair)
add_subdirectory(third_party/uxplay/lib)
set(IMGUI third_party/imgui)
add_library(mirror_ui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_vulkan.cpp src/ui/screen.cpp src/ui/assets.cpp src/platform/video_texture.cpp)
target_include_directories(mirror_ui PUBLIC ${IMGUI} ${IMGUI}/backends src)
target_link_libraries(mirror_ui PUBLIC Vulkan::Vulkan)
add_library(mirror_receiver STATIC src/receiver/airplay.cpp src/receiver/decoder.cpp src/receiver/frame_converter.cpp)
target_include_directories(mirror_receiver PUBLIC src PRIVATE third_party/uxplay/lib)
target_link_libraries(mirror_receiver PUBLIC airplay PkgConfig::AV)
add_library(mirror_media STATIC src/receiver/media_player.cpp src/receiver/dlna.cpp src/receiver/frame_converter.cpp src/platform/audio_queue.cpp)
target_include_directories(mirror_media PUBLIC src)
target_include_directories(mirror_media SYSTEM PRIVATE ${SOUND_INCLUDE_DIRS})
target_link_directories(mirror_media PUBLIC ${SOUND_LIBRARY_DIRS})
target_link_libraries(mirror_media PUBLIC PkgConfig::MEDIA PkgConfig::AV SDL2 ws2_32 iphlpapi)
add_executable(MirrorPSX src/platform/desktop.cpp ${IMGUI}/backends/imgui_impl_glfw.cpp)
target_link_libraries(MirrorPSX PRIVATE mirror_ui mirror_receiver mirror_media glfw)
if(MIRROR_VALIDATE)
    target_compile_definitions(MirrorPSX PRIVATE APP_USE_VULKAN_DEBUG_REPORT)
endif()
enable_testing()
add_executable(decoder_test tests/decoder_test.cpp src/receiver/decoder.cpp src/receiver/frame_converter.cpp)
target_include_directories(decoder_test PRIVATE src)
target_link_libraries(decoder_test PRIVATE PkgConfig::AV)
add_test(NAME decoder COMMAND decoder_test)
add_executable(pipeline_test tests/pipeline_test.cpp src/receiver/frame_converter.cpp)
target_include_directories(pipeline_test PRIVATE src)
target_link_libraries(pipeline_test PRIVATE PkgConfig::AV)
add_test(NAME pipeline COMMAND pipeline_test)
add_executable(ui_test tests/ui_test.cpp)
target_link_libraries(ui_test PRIVATE mirror_ui)
add_test(NAME ui COMMAND ui_test)
set_tests_properties(ui PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
add_executable(receiver_test tests/receiver_test.cpp)
target_link_libraries(receiver_test PRIVATE mirror_receiver)
add_executable(pin_harness tests/pin_harness.cpp)
target_link_libraries(pin_harness PRIVATE mirror_receiver)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/receiver-test")
add_test(NAME receiver COMMAND receiver_test)
add_executable(ps5_random_test tests/ps5_random_test.cpp src/platform/ps5_random.cpp)
target_include_directories(ps5_random_test PRIVATE src)
target_link_libraries(ps5_random_test PRIVATE airplay)
target_compile_options(ps5_random_test PRIVATE -UNDEBUG)
add_test(NAME ps5_random_bridge COMMAND ps5_random_test)
set_tests_properties(receiver PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/receiver-test" TIMEOUT 30)
add_executable(ui_snapshot tools/ui-snapshot.cpp)
target_link_libraries(ui_snapshot PRIVATE mirror_ui)
add_executable(decoder_benchmark tools/decoder-benchmark.cpp)
target_link_libraries(decoder_benchmark PRIVATE mirror_receiver)
add_executable(media_harness tests/media_harness.cpp)
target_link_libraries(media_harness PRIVATE mirror_media)
find_program(FFMPEG_EXECUTABLE NAMES ffmpeg HINTS "C:/FFmpeg/bin")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME pin_pairing COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/pin_integration.py" "$<TARGET_FILE:pin_harness>")
set_tests_properties(pin_pairing PROPERTIES TIMEOUT 20)
if(FFMPEG_EXECUTABLE)
    set(MEDIA_FIXTURE "${CMAKE_BINARY_DIR}/media-fixture.mp4")
    add_custom_command(OUTPUT "${MEDIA_FIXTURE}" COMMAND "${FFMPEG_EXECUTABLE}" -y -hide_banner -loglevel error -f lavfi -i testsrc2=size=480x270:rate=30 -f lavfi -i sine=frequency=440:sample_rate=48000 -t 4 -c:v libx264 -preset ultrafast -g 15 -pix_fmt yuv420p -colorspace bt709 -c:a aac -movflags +faststart "${MEDIA_FIXTURE}")
    add_custom_target(media_fixture ALL DEPENDS "${MEDIA_FIXTURE}")
    add_test(NAME media COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/media_integration.py" "$<TARGET_FILE:media_harness>" "${MEDIA_FIXTURE}")
    set_tests_properties(media PROPERTIES TIMEOUT 35)
endif()
