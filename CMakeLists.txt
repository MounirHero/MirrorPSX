cmake_minimum_required(VERSION 3.22)
project(MirrorPSX VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Vulkan REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
include(FetchContent)
# The initial tree did not ship Dear ImGui. Pin a release so builds are reproducible.
# An externally provided checkout can be used for offline development.
set(MIRRORPSX_IMGUI_DIR "" CACHE PATH "Optional existing Dear ImGui checkout")
if(MIRRORPSX_IMGUI_DIR)
    set(IMGUI "${MIRRORPSX_IMGUI_DIR}")
else()
    FetchContent_Declare(mirror_imgui
        GIT_REPOSITORY https://github.com/ocornut/imgui.git
        GIT_TAG v1.92.9b
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(mirror_imgui)
    set(IMGUI "${mirror_imgui_SOURCE_DIR}")
endif()
if(NOT EXISTS "${IMGUI}/imgui.cpp" OR NOT EXISTS "${IMGUI}/backends/imgui_impl_vulkan.cpp")
    message(FATAL_ERROR "MIRRORPSX_IMGUI_DIR does not contain a Dear ImGui checkout")
endif()
set(NO_MARCH_NATIVE ON)
add_subdirectory(third_party/uxplay/lib/mdnsd)
add_subdirectory(third_party/uxplay/lib/llhttp)
add_subdirectory(third_party/uxplay/lib/playfair)
add_subdirectory(third_party/uxplay/lib)
add_library(mirror_ui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_vulkan.cpp src/ui/screen.cpp src/platform/video_texture.cpp)
target_include_directories(mirror_ui PUBLIC ${IMGUI} ${IMGUI}/backends src)
target_link_libraries(mirror_ui PUBLIC Vulkan::Vulkan)
add_library(mirror_receiver STATIC src/receiver/airplay.cpp src/receiver/decoder.cpp)
target_include_directories(mirror_receiver PUBLIC src PRIVATE third_party/uxplay/lib)
target_link_libraries(mirror_receiver PUBLIC airplay PkgConfig::AV)
add_executable(MirrorPSX src/platform/desktop.cpp ${IMGUI}/backends/imgui_impl_glfw.cpp)
target_link_libraries(MirrorPSX PRIVATE mirror_ui mirror_receiver glfw)
include(CTest)
if(BUILD_TESTING)
    add_executable(decoder_test tests/decoder_test.cpp src/receiver/decoder.cpp)
    target_include_directories(decoder_test PRIVATE src)
    target_link_libraries(decoder_test PRIVATE PkgConfig::AV)
    add_test(NAME decoder COMMAND decoder_test)
    add_executable(state_test tests/state_test.cpp)
    target_include_directories(state_test PRIVATE src)
    add_test(NAME state COMMAND state_test)
endif()
